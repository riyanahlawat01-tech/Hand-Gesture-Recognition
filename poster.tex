\seccard{Abstract}{
\small
A complete hand gesture recognition system built \textbf{entirely from scratch} using a custom CNN --- no pretrained weights, transfer learning, or external APIs. Recognises \textbf{32 ASL gestures} achieving \highlight{99.24\% best validation accuracy}. Dataset of 12,800 images self-collected via custom webcam pipeline. Runs at \textbf{14--18 FPS on CPU}.\\[4pt]
\textbf{\textcolor{upesblue}{Keywords:}} CNN\enspace$\cdot$\enspace ASL\enspace$\cdot$\enspace Custom Dataset\enspace$\cdot$\enspace OpenCV\enspace$\cdot$\enspace TensorFlow
}

\vspace{5pt}

\seccard{Dataset}{
\begin{center}
\includegraphics[width=0.92\linewidth,trim=0 100 0 0,clip]{dataset.png}
\end{center}
\vspace{2pt}
\scriptsize
\begin{itemize}\setlength\itemsep{2pt}
  \hitem{32 classes:} A--Z + hello, bye, yes, no, thankyou, perfect
  \hitem{400 images/class:} 200 left + 200 right hand
  \hitem{12,800 total:} 80\% train / 20\% val
  \item ROI-box --- only hand region saved, no face
  \item All images captured by \textbf{Riyan Ahlawat}
\end{itemize}
}

\vspace{5pt}

\seccard{Methodology}{
\scriptsize
\begin{enumerate}\setlength\itemsep{2pt}
  \item \textbf{Collect:} ROI webcam script; left + right hand
  \item \textbf{Preprocess:} Resize 64$\times$64; normalise [0,1]; tf.data
  \item \textbf{Augment:} Flip, brightness, contrast, hue
  \item \textbf{Model:} 4 conv blocks 32$\to$64$\to$128$\to$256; GAP; Dense
  \item \textbf{Train:} Adam + ReduceLROnPlateau + EarlyStopping
  \item \textbf{Infer:} OpenCV; every 2 frames; top-3 overlay
\end{enumerate}
}

\vspace{5pt}

\seccard{Challenges}{
\scriptsize
\begin{itemize}\setlength\itemsep{2pt}
  \hitem{Python 3.14 ---} TF needs $\le$3.12; installed 3.11
  \hitem{Domain gap ---} black-bg dataset failed; rebuilt from webcam
  \hitem{GPU on Windows ---} TF $\ge$2.11 no native GPU; used CPU
  \hitem{Laptop shutdown ---} sleep disabled; charger connected
  \hitem{Save format ---} .keras incompatible TF2.13; used .h5
\end{itemize}
}

\seccard{Training Configuration}{
\scriptsize
\begin{itemize}\setlength\itemsep{2pt}
  \item tf.data pipeline with prefetching
  \item Data augmentation: random flip, brightness ($\pm$0.2), contrast ($\pm$0.2), hue ($\pm$0.05)
  \item He Normal weight initialisation
  \item BatchNorm after every Conv and Dense layer
  \item Global Average Pooling replaces Flatten to reduce params
  \item ReduceLROnPlateau: monitor val\_loss, patience 5
  \item EarlyStopping: monitor val\_accuracy, patience 10
  \item ModelCheckpoint: saves best .h5 weights
\end{itemize}
}

\seccard{Model Architecture Summary}{
\scriptsize
\begin{itemize}\setlength\itemsep{2pt}
  \item \textbf{Conv Block 1:} Conv2D(32) $\to$ BN $\to$ ReLU $\to$ MaxPool $\Rightarrow$ 32$\times$32$\times$32
  \item \textbf{Conv Block 2:} Conv2D(64) $\to$ BN $\to$ ReLU $\to$ MaxPool $\Rightarrow$ 16$\times$16$\times$64
  \item \textbf{Conv Block 3:} Conv2D(128) $\to$ BN $\to$ ReLU $\to$ MaxPool $\Rightarrow$ 8$\times$8$\times$128
  \item \textbf{Conv Block 4:} Conv2D(256) $\to$ BN $\to$ ReLU $\to$ MaxPool $\Rightarrow$ 4$\times$4$\times$256
  \item \textbf{GAP} $\Rightarrow$ (256,)
  \item \textbf{Dense(512)} + BN + ReLU + Dropout(0.5)
  \item \textbf{Dense(256)} + BN + ReLU + Dropout(0.3)
  \item \textbf{Dense(32)} + Softmax $\Rightarrow$ Output
\end{itemize}
